def rep(p, pares):
    s = io.open(p, encoding="utf-8").read()
    for a, b in pares:
        assert s.count(a) == 1, (p, a[:70], s.count(a))
        s = s.replace(a, b)
    io.open(p, "w", encoding="utf-8", newline="\n").write(s)
